import { runMcpServer, stopMcpServer } from "@/mcp-server.js";
import {
  getMcpPid,
  isMcpServerRunning,
  killAllMcpProcesses,
  listMcpProcesses,
} from "@/utils/mcp-process-utils.js";

import { BaseCommand } from "./BaseCommand.js";

export class McpCommand extends BaseCommand {
  public name = "mcp";
  public description = `Démarre, stoppe ou relance le serveur MCP pour interagir avec Claude ou un IDE

Commandes disponibles:
  mclp mcp start      # Démarre le serveur (avec vérification)
  mclp mcp stop       # Arrête proprement le serveur
  mclp mcp restart    # Redémarre le serveur
  mclp mcp status     # Affiche le statut + PID
  mclp mcp ps         # Liste tous les processus MCP
  mclp mcp kill       # Force l'arrêt (si bloqué)`;
  public arguments = "<action>";

  async execute(args: string[]): Promise<void> {
    const [action] = args;

    if (!action) {
      console.log("❌ Action manquante");
      console.log(this.description);
      return;
    }

    switch (action) {
      case "start":
        await this.handleStart();
        break;
      case "stop":
        await this.handleStop();
        break;
      case "restart":
        await this.handleRestart();
        break;
      case "status":
        await this.handleStatus();
        break;
      case "ps":
        await this.handlePs();
        break;
      case "kill":
        await this.handleKill();
        break;
      default:
        console.log(`❌ Action inconnue: ${action}`);
        console.log(this.description);
    }
  }

  private async handleStart(): Promise<void> {
    const isRunning = await isMcpServerRunning();

    if (isRunning) {
      console.log("⚠️ Un serveur MCP est déjà en cours d'exécution");
      console.log("💡 Utilisez 'mclp mcp stop' pour l'arrêter d'abord");
      console.log("💡 Ou utilisez 'mclp mcp restart' pour redémarrer");
      return;
    }

    console.log("🚀 Démarrage du serveur MCP...");
    await runMcpServer();
  }

  private async handleStop(): Promise<void> {
    console.log("🛑 Arrêt du serveur MCP...");
    await stopMcpServer();
    await killAllMcpProcesses();
    console.log("✅ Serveur MCP arrêté");
  }

  private async handleRestart(): Promise<void> {
    console.log("🔄 Redémarrage du serveur MCP...");
    await stopMcpServer();
    await killAllMcpProcesses();

    // Attendre un peu avant de redémarrer
    await new Promise((resolve) => setTimeout(resolve, 1000));

    console.log("🚀 Démarrage du serveur MCP...");
    await runMcpServer();
  }

  private async handleStatus(): Promise<void> {
    const isRunning = await isMcpServerRunning();
    const pid = await getMcpPid();

    console.log("\n📊 Statut du serveur MCP:");
    console.log("─".repeat(40));

    if (isRunning && pid) {
      console.log("✅ État: En cours d'exécution");
      console.log(`📌 PID: ${pid}`);
    } else {
      console.log("❌ État: Arrêté");
    }

    console.log("─".repeat(40) + "\n");
  }

  private async handlePs(): Promise<void> {
    console.log("🔍 Recherche de processus MCP...\n");
    const processes = await listMcpProcesses();

    if (processes.length === 0) {
      console.log("❌ Aucun processus MCP trouvé");
    } else {
      console.log(`✅ ${processes.length} processus trouvé(s):\n`);
      console.log("─".repeat(80));
      processes.forEach((proc, index) => {
        console.log(`${index + 1}. ${proc}`);
      });
      console.log("─".repeat(80));
    }
  }

  private async handleKill(): Promise<void> {
    console.log("💀 Arrêt forcé de tous les processus MCP...");
    await killAllMcpProcesses();
    console.log("✅ Tous les processus MCP ont été arrêtés");
  }
}
